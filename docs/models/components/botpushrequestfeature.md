# BotPushRequestFeature

## Example Usage

```typescript
import { BotPushRequestFeature } from "@steamsets/client-ts/models/components";

let value: BotPushRequestFeature = {
  activationCommand: "!useinv",
  commands: [
    "!check",
    "!buytf",
  ],
  feature: "restockAlerts",
};
```
